











 
 

  
 
CREATE VIEW [dbo].[vw_ap_type0_dtl]  
AS
select 1 as system_type,ic.am_mode_id
	,isnull(c.i_is_monthly,0) as i_is_monthly
	, e.sp_check_period_hdr_id
	, e.c_code as c_contract, e.c_code as c_checking_code ,e.d_checking_date
	, ed.dc_bg_budget_type_id
	, ct.dc_cost_acc_id
	, ct.dc_cost_id
	, CASE
	 WHEN YEAR(e.d_checking_date) = (SELECT TOP 1 i_yyyy FROM NMU_ERP.dbo.sp_tor WHERE tor_id=c.sp_tor_id) THEN 1
	 WHEN YEAR(e.d_checking_date) > (SELECT TOP 1 i_yyyy FROM NMU_ERP.dbo.sp_tor WHERE tor_id=c.sp_tor_id) THEN 2 
	 ELSE 9
	END as i_type_year
	,  (SELECT TOP 1 i_yyyy FROM NMU_ERP.dbo.sp_tor WHERE tor_id=c.sp_tor_id) as c_budget_year

	, case isnull(i_is_product_last,0) when 0 then 'ที่ไม่ได้ของ'
		when 1 then 'วัสดุ'
		when 2 then 'สินทรัพทย์'
		else 'ไม่ทราบ' end as c_product_type
		 

	,CASE WHEN (SELECT COUNT(sa.tor_id) FROM EIS_PROCURE..sp_tor sa
					INNER JOIN EIS_PROCURE..sp_tor_i_work_type sb ON sa.i_working_type = sb.i_work_type_id
				WHERE sb.i_work_type_id IN (6, 7, 8) AND sa.tor_id = t.tor_id) > 0 THEN 
		(SELECT dc_acc_id FROM NMU_DATACENTER.dbo.dc_acc 
		WHERE i_enable = 1 AND i_delete = 2 AND c_code = '20101040101')  --เจ้าหนี้งานก่อสร้าง  @25-9-68
	ELSE
		(select top 1 dc_acc_id from NMU_ERP.dbo.sp_acc_inv where i_enabled = 1)  --เจ้าหนี้ค่าสินค้าและบริการ
	END  as dc_acc_cr_id 

	, case isnull(h.i_is_product_last,0) 
		when 0 then 
			(select top 1 cc.dc_acc_id
			   from NMU_DATACENTER..dc_bg_acc_hdr  aa
						INNER JOIN NMU_DATACENTER..dc_bg_acc_dtl bb on aa.dc_bg_acc_hdr_id = bb.dc_bg_acc_hdr_id  
						INNER JOIN NMU_DATACENTER..dc_acc cc on cc.dc_acc_id = bb.dc_acc_id  
						where aa.i_enable = 1 and bb.i_enable = 1 and  cc.i_enable =1 and cc.i_last = 1 and cc.i_group in(1,5)   and  aa.bg_expense_id = ed.po_expense_id)
		when 1 then  
			case when (ed.po_expense_id=415) then (select top 1 tt.dc_acc_id from NMU_DATACENTER.dbo.dc_acc tt where tt.c_code = '50301050120' and tt.i_enable=1) else (select top 1 dc_acc_id from NMU_DATACENTER.dbo.inv_mode_acc where inv_mode_id = ic.inv_mode_id) end
		when 2 then 
			(select top 1 dc_acc4_id from NMU_DATACENTER.dbo.am_mode_acc where am_mode_id = ic.am_mode_id)
		else 0 end dc_acc_dr_id /*dc_acc_id ของรายการซื้อ*/
	--, ic.f_net_total_price
		--, ic.f_net_total_price
	, CASE 
      WHEN ic.i_workin_process = 1 
           THEN ic.f_wip_total_price 
      ELSE ic.f_net_total_price 
  END AS f_net_total_price
	,isnull((select  1 from FM_NMU..gl_sp_map_bg_cost where i_enable=1 and dc_expense_budget_type_id_nmu = ed.dc_bg_budget_type_id),0)  as i_is_fm
	,isnull(h.i_is_product_last,0) as i_is_product_last
		,ISNULL((select top 1 ff.dc_cost_id from FM_NMU..gl_sp_map_bg_cost ff where ff.dc_expense_budget_type_id_nmu = h.dc_expense_budget_type_id and ff.i_enable=1  ),0)  as dc_cost_id_fm 

 from NMU_ERP.dbo.sp_check_period_hdr e 
 inner join NMU_ERP.dbo.sp_check_period_dtl ed on ed.sp_check_period_hdr_id =e.sp_check_period_hdr_id
 inner join NMU_ERP.dbo.sp_tor_contract c on c.sp_tor_contract_id=e.sp_tor_contract_id  and c.i_enabled=1 
 inner join NMU_ERP.dbo.sp_tor t on t.tor_id=c.sp_tor_id   and t.i_enabled=1 
 inner join NMU_DATACENTER.dbo.dc_cost ct on ct.dc_cost_id=t.dc_cost_id
 inner join NMU_ERP.dbo.sp_tor_hdr_period h on h.sp_tor_hdr_period_id=e.sp_tor_hdr_period_id and h.i_enabled=1 
 inner join NMU_ERP.dbo.sp_tranf_item ic on ic.sp_check_period_hdr_id=e.sp_check_period_hdr_id and ic.i_enabled=1 
 
 where  isnull(e.c_code,'') !='' 
		--  and case when i_is_product_last in(1, 2) and isnull(h.i_is_last,0)= 0 then 1 
		--  when isnull(i_is_product_last,0) =0 then 1 else 0 end =1
        and isnull(e.i_is_register,0) = 0
        and case when isnull(ed.i_product_type,0) in (2)   and isnull(h.i_is_last,0) = 0 then 1 
        when isnull(ed.i_product_type,0) in (0,1) then 1 
        else 0  end  = 1
		 and NOT EXISTS (SELECT 1 FROM (VALUES ('AP') , ('CH')  ) AS patterns(p) WHERE left(e.c_code,2) = p )
 
---------------------------------------------
UNION ALL
---------------------------------------------

select 3 as system_type,ic.am_mode_id 
,isnull(c.i_is_monthly,0) as i_is_monthly
	, e.sp_check_period_hdr_id
	, e.c_code as c_contract, e.c_code as c_checking_code ,e.d_checking_date
	, ed.dc_bg_budget_type_id
	, ct.dc_cost_acc_id
	, ct.dc_cost_id
	, CASE
	 WHEN YEAR(e.d_checking_date) = (SELECT TOP 1 i_yyyy FROM EIS_PROCURE.dbo.sp_tor WHERE tor_id=c.sp_tor_id) THEN 1
	 WHEN YEAR(e.d_checking_date) > (SELECT TOP 1 i_yyyy FROM EIS_PROCURE.dbo.sp_tor WHERE tor_id=c.sp_tor_id) THEN 2 
	 ELSE 9
	END as i_type_year
	,  (SELECT TOP 1 i_yyyy FROM EIS_PROCURE.dbo.sp_tor WHERE tor_id=c.sp_tor_id) as c_budget_year

	, case isnull(h.i_is_product_last,0) when 0 then 'ที่ไม่ได้ของ'
		when 1 then 'วัสดุ'
		when 2 then 'สินทรัพทย์'
		else 'ไม่ทราบ' end as c_product_type

	--, (select top 1 dc_acc_id from EIS_PROCURE.dbo.sp_acc_inv) as dc_acc_cr_id /*dc_acc_id เจ้าหนี้ค่าสินค้าและบริการ*/

	,CASE WHEN (SELECT COUNT(sa.tor_id) FROM EIS_PROCURE..sp_tor sa
					INNER JOIN EIS_PROCURE..sp_tor_i_work_type sb ON sa.i_working_type = sb.i_work_type_id
				WHERE sb.i_work_type_id IN (6, 7, 8) AND sa.tor_id = t.tor_id) > 0 THEN 
		(SELECT dc_acc_id FROM NMU_DATACENTER.dbo.dc_acc 
		WHERE i_enable = 1 AND i_delete = 2 AND c_code = '20101040101')  --เจ้าหนี้งานก่อสร้าง    @25-9-68
	ELSE
		(select top 1 dc_acc_id from EIS_PROCURE.dbo.sp_acc_inv where i_enabled = 1)  --เจ้าหนี้ค่าสินค้าและบริการ
	END  as dc_acc_cr_id 

	, case isnull(i_is_product_last,0) 
		when 0 then 
			(select top 1 cc.dc_acc_id
			   from NMU_DATACENTER..dc_bg_acc_hdr  aa
						INNER JOIN NMU_DATACENTER..dc_bg_acc_dtl bb on aa.dc_bg_acc_hdr_id = bb.dc_bg_acc_hdr_id  
						INNER JOIN NMU_DATACENTER..dc_acc cc on cc.dc_acc_id = bb.dc_acc_id  
						where aa.i_enable = 1 and bb.i_enable = 1 and  cc.i_enable =1 and cc.i_last = 1 and cc.i_group in(1,5) and  aa.bg_expense_id = ed.po_expense_id)
		when 1 then 
			(select top 1 dc_acc1_id from NMU_DATACENTER.dbo.inv_mode_acc where inv_mode_id = ic.inv_mode_id)
		when 2 then 
			(select top 1 dc_acc4_id from NMU_DATACENTER.dbo.am_mode_acc where am_mode_id = ic.am_mode_id)
		else 0 end dc_acc_dr_id /*dc_acc_id ของรายการซื้อ*/
	--, ic.f_net_total_price 
	, CASE 
      WHEN ic.i_workin_process = 1 
           THEN ic.f_wip_total_price 
      ELSE ic.f_net_total_price 
  END AS f_net_total_price
	,isnull((select  1 from FM_NMU..gl_sp_map_bg_cost where i_enable=1 and dc_expense_budget_type_id_nmu = ed.dc_bg_budget_type_id),0)  as i_is_fm
	,isnull(h.i_is_product_last,0) as i_is_product_last
		,ISNULL((select top 1 ff.dc_cost_id from FM_NMU..gl_sp_map_bg_cost ff where ff.dc_expense_budget_type_id_nmu = h.dc_expense_budget_type_id and ff.i_enable=1  ),0)  as dc_cost_id_fm 

 from EIS_PROCURE.dbo.sp_check_period_hdr e 
 inner join EIS_PROCURE.dbo.sp_check_period_dtl ed on ed.sp_check_period_hdr_id =e.sp_check_period_hdr_id
 inner join EIS_PROCURE.dbo.sp_tor_contract c on c.sp_tor_contract_id=e.sp_tor_contract_id  and c.i_enabled=1 
 inner join EIS_PROCURE.dbo.sp_tor t on t.tor_id=c.sp_tor_id   and t.i_enabled=1 
 inner join NMU_DATACENTER.dbo.dc_cost ct on ct.dc_cost_id=t.dc_cost2_id
 inner join EIS_PROCURE.dbo.sp_tor_hdr_period h on h.sp_tor_hdr_period_id=e.sp_tor_hdr_period_id and h.i_enabled=1 
 inner join EIS_PROCURE.dbo.sp_tranf_item ic on ic.sp_check_period_hdr_id=e.sp_check_period_hdr_id and ic.i_enabled=1  
 where  isnull(e.c_code,'') !='' 
		--  and case when i_is_product_last in(1, 2) and isnull(h.i_is_last,0)= 0 then 1 
		--  when isnull(i_is_product_last,0) =0 then 1 else 0 end =1
		and isnull(e.i_is_register,0) = 0
        and case when isnull(ed.i_product_type,0) in (2)   and isnull(h.i_is_last,0) = 0 then 1 
        when isnull(ed.i_product_type,0) in (0,1) then 1 
        else 0  end  = 1
		and NOT EXISTS (SELECT 1 FROM (VALUES ('AP') , ('CH')  ) AS patterns(p) WHERE left(e.c_code,2) = p )
